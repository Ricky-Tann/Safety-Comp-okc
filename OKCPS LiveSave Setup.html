OKCPS Live Save Setup

Creates the live_answers table for real-time inspection saving.

Go to your Supabase SQL Editor and run this SQL:

-- Run this in Supabase SQL Editor -- supabase.com/dashboard/project/dkmmrdukdejvhuwptaug CREATE TABLE IF NOT EXISTS live_answers ( session_id TEXT NOT NULL, item_key TEXT NOT NULL, building TEXT, insp_date TEXT, inspector TEXT, status TEXT, note TEXT, ca_required BOOLEAN DEFAULT FALSE, ca_due TEXT, photos_json TEXT, updated_at TIMESTAMPTZ DEFAULT NOW(), PRIMARY KEY (session_id, item_key) ); CREATE TABLE IF NOT EXISTS live_meta ( session_id TEXT PRIMARY KEY, building TEXT, insp_date TEXT, inspector TEXT, contact TEXT, insp_type TEXT, weather TEXT, notes TEXT, cond_json TEXT, nums_json TEXT, dates_json TEXT, hazards_json TEXT, updated_at TIMESTAMPTZ DEFAULT NOW() ); ALTER TABLE live_answers DISABLE ROW LEVEL SECURITY; ALTER TABLE live_meta DISABLE ROW LEVEL SECURITY; GRANT ALL ON live_answers TO anon, authenticated; GRANT ALL ON live_meta TO anon, authenticated;
